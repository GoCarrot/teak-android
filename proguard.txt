### Billing, reached by name or reflection ###
# Billing ships no keep rules for these, so a minified host app would otherwise rename or strip
# them. The product-details listener goes through a dynamic proxy (all billing versions); the
# QueryProductDetailsResult unwrap (8+), the billing version and the PendingPurchasesParams probe
# go through reflection.
-keep interface com.android.billingclient.api.ProductDetailsResponseListener { *; }
-keep class com.android.billingclient.api.QueryProductDetailsResult { *; }
-keep class com.android.billingclient.BuildConfig { *; }
-keepnames class com.android.billingclient.api.PendingPurchasesParams
